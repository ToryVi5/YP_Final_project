# SQL запрос
SELECT c.login, COUNT(o.id) AS "deliveryCount"
 FROM "Couriers" AS c
 LEFT JOIN "Orders" AS o ON c.id = o."courierId" AND o."inDelivery" = true
 GROUP BY c.login;

# Пояснение:
# SELECT c. login, COUNT(o.id) AS "deliveryCount": выводит логин курьера и количество заказов, принятых этим курьером;
# FROM "Couriers" AS c: берёт данные из таблицы Couriers (курьеров) - для удобства сократили до AS c;
# LEFT JOIN "Orders" AS o ON c.id = o."courierId" AND o."inDelivery" = true: 
# Первое предложение до AND - Объединяет обе таблицы Couriers и Orders по идентификатору курьера
# Второе предложение после AND - Чтобы отобразить курьера с нулевым количеством, условие o."inDelivery" = true переносим в часть ON
# GROUP BY c.login; - группируем результат по логинам.
